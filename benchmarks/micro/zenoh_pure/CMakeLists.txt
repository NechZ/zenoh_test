cmake_minimum_required(VERSION 3.16)
project(zenoh_pure_bench CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# zenoh-c / zenoh-cpp as shipped with ros-jazzy-zenoh-cpp-vendor (same Zenoh as rmw_zenoh)
set(ZENOH_ROOT /opt/ros/jazzy/opt/zenoh_cpp_vendor CACHE PATH "zenoh-c/zenoh-cpp install prefix")

foreach(t zpub zsub)
  add_executable(${t} ${t}.cpp)
  target_compile_definitions(${t} PRIVATE ZENOHCXX_ZENOHC)
  target_include_directories(${t} PRIVATE ${ZENOH_ROOT}/include)
  target_link_directories(${t} PRIVATE ${ZENOH_ROOT}/lib)
  target_link_libraries(${t} PRIVATE zenohc pthread)
  set_target_properties(${t} PROPERTIES BUILD_RPATH ${ZENOH_ROOT}/lib)
endforeach()
